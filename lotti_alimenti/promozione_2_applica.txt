# PROMOZIONE - APPLICA: SCRIVE DAVVERO sul database (da bozza a verificato).
# Promuove a "verificato" SOLO gli alimenti che hanno superato tutti i controlli automatici (gli altri restano bozza).
# Lancialo SOLO dopo aver fatto la prova e dopo che Claude ti ha detto che il risultato e' a posto.
# Durata: circa 20-30 minuti (1 secondo di pausa per alimento + migliaia di scritture). NON chiudere PowerShell.
# Copia TUTTO il file (Ctrl+A, Ctrl+C), incollalo in PowerShell aperto nella cartella del progetto, aspetta "Finito".
New-Item -ItemType Directory -Force lotti_alimenti | Out-Null
$f = "lotti_alimenti\esito_promozione_applica.txt"
Remove-Item $f -ErrorAction SilentlyContinue
"PROMOZIONE APPLICA - " + (Get-Date) | Out-File $f -Encoding utf8
npm run verify:golden-set -- --apply 2>&1 | Out-File $f -Append -Encoding utf8
"FINE - " + (Get-Date) | Out-File $f -Append -Encoding utf8
Write-Host ""
Write-Host "Finito la promozione. Riepilogo:"
Get-Content $f | Select-String -Pattern "Righe draft|MATCH:|MISMATCH:|BLOCKED|NOT_FOUND|UNIT_MISMATCH|FOOD_UNREACHABLE|NOT_MAPPED|promosse|promossi|Errore fatale"
